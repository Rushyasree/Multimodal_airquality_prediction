\begin{frame}{1. Motivation \& Urban Air Pollution Crisis in India}
  \begin{itemize}
    \item \textbf{Public Health Emergency}: Severe ambient particulate matter ($\text{PM}_{2.5}$ and $\text{PM}_{10}$) regularly exceeds NAAQS ($60/100\ \mu\text{g/m}^3$) and WHO thresholds ($15/45\ \mu\text{g/m}^3$) by $5\times$--$10\times$.
    \item \textbf{Extreme Seasonal Volatility}: Post-monsoon agricultural stubble burning and winter thermal boundary layer inversions cause episodic smog shocks.
    \item \textbf{Operational Need}: Municipal authorities require reliable 1-hour ahead ($t+1$) early warnings for dynamic traffic management, school advisories, and vulnerable population protection.
  \end{itemize}
\end{frame}

\begin{frame}{2. Three Operational Barriers in Indian Sensor Networks}
  \begin{block}{1. Telemetry Dropouts \& Gaseous Sensor Downtime}
    Standard 27-channel deep LSTMs require trace gases ($\text{NO}_x, \text{CO}, \text{SO}_2, \text{O}_3$). In real CAAQMS operations, secondary analyzers experience frequent maintenance dropouts, causing high-dimensional models to crash.
  \end{block}
  \begin{block}{2. Uncalibrated Uncertainty in Heavy-Tailed Regimes}
    Particulate residuals during episodic events exhibit non-Gaussian, heteroskedastic distributions. Standard Gaussian confidence intervals fail.
  \end{block}
  \begin{block}{3. Unverified Multimodal Hypotheses}
    Recent literature proposes fusing opportunistic outdoor camera photos with ground sensors without auditing spatial-temporal alignment.
  \end{block}
\end{frame}

\begin{frame}{3. Dataset Scope: Historical Reanalysis \& Nationwide Live Cohort}
  \begin{itemize}
    \item \textbf{Historical Regulatory Dataset}: 1,369,779 hourly records from 90 CPCB continuous stations (2015--2020) across 23 States and Union Territories.
    \item \textbf{Meteorological Reanalysis}: Hourly ERA5-Land variables ($T, T_d, RH, u, v, P$) bilinearly matched to all 90 station coordinates.
    \item \textbf{Nationwide Longitudinal Live Portfolio}: 15 continuous CAAQMS stations across 11 Indian States/UTs evaluated over 236,289 live sequences and 506,848 seasonal evaluations.
    \item \textbf{Environmental Vision}: Curated outdoor photographs from PM25Vision across Indian coordinates.
  \end{itemize}
\end{frame}

\begin{frame}{5. Empirical Negative Result: Opportunistic Multimodal Sensing}
  \textbf{Auditing 640 unique outdoor images from PM25Vision paired with CPCB monitors:}
  \begin{enumerate}
    \item \textbf{Spatial Displacement}: 50.6\% of images exceed 5 km from the nearest monitor (mean: 92.8 km).
    \item \textbf{Temporal Mismatch}: Images have only calendar dates (`YYYY-MM-DD`) with zero sub-daily timestamps, despite a \textbf{166.72 $\mu\text{g/m}^3$} average diurnal station $\text{PM}_{2.5}$ swing.
    \item \textbf{Burst Clustering}: Matched cohort represents only \textbf{36 independent groups} (40\% from 2 days).
    \item \textbf{Label Discrepancy}: Images exhibit a systematic \textbf{$+70.74\ \mu\text{g/m}^3$} positive calibration bias.
    \item \textbf{Controlled Ablation Failure}: Multimodal early fusion degraded MAE by \textbf{12.19 $\mu\text{g/m}^3$} (85.86 vs 73.67 $\mu\text{g/m}^3$, $R^2 = -0.5064$).
  \end{enumerate}
\end{frame}

\begin{frame}{6. Standalone Adapted GBDT Architecture}
  \textbf{Canonical 49-Feature Autoregressive Schema (Zero Trace Gas Reliance):}
  \begin{itemize}
    \item \textbf{Particulate Lags (24)}: $\text{PM}_{2.5}(t-k), \text{PM}_{10}(t-k)$ for $k \in \{0, \dots, 11\}$.
    \item \textbf{Rolling Statistics (14)}: 3h, 6h, 12h, 24h rolling mean, std, min, max, coarse-to-fine ratio.
    \item \textbf{Cyclical Harmonics (4)}: $\sin/\cos$ of hour-of-day and month-of-year.
    \item \textbf{Calendar (3)}: Day of week, day of year, weekend flag.
    \item \textbf{Spatial Coordinates (4)}: Latitude, longitude, elevation, target-encoded station ID.
  \end{itemize}
  \begin{alertblock}{Zero Trace Gas Failure Mode}
    Operates strictly on particulate telemetry, guaranteeing 100\% uptime during trace gas outages.
  \end{alertblock}
\end{frame}

\begin{frame}{7. Severity-Gated Adaptive Ensemble}
  \textbf{Dynamic Routing Based on Instantaneous Concentration Severity:}
  $$\hat{y}(t+1) = \alpha(C_t) \hat{y}_{\text{GBDT}}(t+1) + (1 - \alpha(C_t)) \hat{y}_{\text{LSTM}}(t+1)$$
  \begin{itemize}
    \item $C_t = \max(\text{PM}_{2.5}(t), \text{PM}_{10}(t))$ monitors real-time pollution level.
    \item In clean/moderate air ($C_t < 90\ \mu\text{g/m}^3$), deep sequence representations (LSTM) dominate.
    \item In hazardous smog shocks ($C_t > 250\ \mu\text{g/m}^3$), GBDT receives primary weight, preventing recurrent activation saturation.
  \end{itemize}
\end{frame}

\begin{frame}{10. Rigorous Statistical Hypothesis Testing}
  \textbf{Stationary Block Bootstrap ($B=5,000$, block length $L=24\text{h}$ to preserve diurnal correlation):}
  \begin{itemize}
    \item \textbf{Pooled $\text{PM}_{2.5}$ Nationwide} ($N=236,289$):
      $$\Delta = -0.28\ \mu\text{g/m}^3,\quad 95\%\ \text{CI}: [-0.30, -0.25],\quad p_{\text{Wilcoxon}} = 4.98 \times 10^{-12}$$
    \item \textbf{Pooled $\text{PM}_{10}$ Nationwide} ($N=236,289$):
      $$\Delta = -0.87\ \mu\text{g/m}^3,\quad 95\%\ \text{CI}: [-0.94, -0.79],\quad p_{\text{Wilcoxon}} = 7.57 \times 10^{-24}$$
    \item \textbf{Multiple Testing Controls}: All 18 pre-registered tests pass Bonferroni FWER ($p < 0.0028$) and Benjamini-Hochberg FDR ($q < 0.05$).
    \item \textbf{Severe Hazard Regimes}: In *Very Poor* air ($N=4,847$), GBDT achieves a \textbf{2.48 $\mu\text{g/m}^3$ MAE reduction} ($p = 3.71 \times 10^{-25}$).
  \end{itemize}
\end{frame}

\begin{frame}{13. Conclusion \& Research Contributions}
  \begin{enumerate}
    \item \textbf{Operational Standalone Architecture}: Built a 49-variable standalone GBDT that eliminates failure modes caused by trace gas dropouts, beating persistence by up to 5.92\%.
    \item \textbf{Distribution-Free Uncertainty Bounds}: Provided split-conformal calibration with guaranteed finite-sample coverage ($\ge 92.8\%$) under severe smog conditions.
    \item \textbf{Physical Transport Demarcation}: Isolated coastal sea-breeze advective gains from inland local emission dominance.
    \item \textbf{Methodological Honesty}: Established the empirical negative result on opportunistic environmental imagery, defining the boundary conditions for future synchronous sensing.
  \end{enumerate}
